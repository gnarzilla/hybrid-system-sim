# GPIO Operations
WRITE_GPIO_PIN 18 HIGH
READ_GPIO_PIN 17
WRITE_GPIO_PIN 18 LOW
READ_GPIO_PIN 17

# Analog Operations with Plotting
AMPLIFY A1 1.2
AMPLIFY A1 1.5
FREQUENCY_MODULATE A1 1.1
AMPLIFY A1 0.8
A2D A1 R2  # Convert the analog value in A1 to digital in R2
READ_ANALOG A1  # Read the current analog value from A1 and add it to plot

# Voltage Check Feedback
CHECK_VOLTAGE A1 1.5  # Check if the analog value in A1 is above 1.5

# Circuit Creation for ngspice
SET_CIRCUIT RESISTOR R1 1k
SET_CIRCUIT CAPACITOR C1 1uF
SET_CIRCUIT VOLTAGE_SOURCE V1 5V
